cmake_minimum_required(VERSION 3.20)
project(FileOpRace LANGUAGES CXX)

add_executable(FileOpRace FileOpRace.cpp)
target_compile_features(FileOpRace PRIVATE cxx_std_17)
target_compile_definitions(FileOpRace PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
target_link_libraries(FileOpRace PRIVATE advapi32)

if(MSVC)
    set_property(TARGET FileOpRace PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(FileOpRace PRIVATE /W4)
else()
    target_compile_options(FileOpRace PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(FileOpRace PRIVATE -municode -static)
endif()
